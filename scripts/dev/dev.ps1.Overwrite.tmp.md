# scripts/dev/dev.ps1 - Development controller for SMKD project (read‑only preflight checks and placeholders for other commands).`n`nparam(    `n    [Parameter(Mandatory=$true, Position=0)][ValidateSet('doctor','start','status','test','diagnose','cache','stop')]`n    [string]$Command,`n    [string]$Scope,`n    [switch]$LastRun`n)`n`nfunction Invoke-Doctor {`n    Write-Output 'Running environment doctor...'`n    $results = @()`n    # Helper to emit result`n    function Add-Result($status, $message) {`n        $results += @{Status=$status;Message=$message}`n    }`n    # 1. Git repository root`n    try {`n        $gitRoot = git rev-parse --show-toplevel 2>$null`n        if ($LASTEXITCODE -ne 0) {throw}`n        Add-Result '[PASS]' ( 'Git repository root: ' + $gitRoot )`n    } catch { Add-Result '[FAIL]' 'Git not found or not a repository' }`n    # 2. Git status (uncommitted changes)`n    try {`n        $gitStatus = git status --porcelain`n        if ($gitStatus) { Add-Result '[WARN]' 'Uncommitted changes present' } else { Add-Result '[PASS]' 'Clean working tree' }`n    } catch { Add-Result '[FAIL]' 'Unable to run git status' }`n    # 3. Submodule status`n    try {`n        $submodule = git submodule status 2>$null`n        if ($submodule -match '(^-|[[:space:]])') { Add-Result '[WARN]' 'Submodule not initialized or out‑of‑date' } else { Add-Result '[PASS]' 'Submodules up‑to‑date' }`n    } catch { Add-Result '[FAIL]' 'Git submodule command failed' }`n    # 4. package.json syntax`n    $pkgPath = Join-Path -Path $PSScriptRoot -ChildPath '..' -Resolve | Join-Path -ChildPath 'package.json'`n    if (Test-Path $pkgPath) {`n        try {`n            Get-Content $pkgPath -Raw | ConvertFrom-Json | Out-Null`n            Add-Result '[PASS]' 'package.json parses as valid JSON'`n        } catch { Add-Result '[FAIL]' 'package.json contains invalid JSON' }`n    } else { Add-Result '[FAIL]' 'package.json not found' }`n    # 5. PNPM workspace validation (dry‑run install)`n    try {`n        $pnpm = Get-Command pnpm -ErrorAction Stop`n        $proc = Start-Process -FilePath $pnpm.Source -ArgumentList 'install','--dry-run' -NoNewWindow -RedirectStandardError $null -RedirectStandardOutput $null -PassThru -Wait -ErrorAction Stop`n        if ($proc.ExitCode -eq 0) { Add-Result '[PASS]' 'PNPM workspace resolves' } else { Add-Result '[FAIL]' 'PNPM workspace dependency resolution failed' }`n    } catch { Add-Result '[FAIL]' 'PNPM not installed or command failed' }`n    # 6. Python interpreter`n    try {`n        $python = Get-Command python -ErrorAction Stop`n        $ver = & $python.Source -c 'import sys, json; print(sys.version)'`n        Add-Result '[PASS]' ('Python interpreter found: ' + $ver.Trim())`n    } catch { Add-Result '[FAIL]' 'Python interpreter not found' }`n    # 7. Required Python packages (fastapi as example)`n    try {`n        & $python.Source -c 'import fastapi' 2>$null`n        if ($LASTEXITCODE -eq 0) { Add-Result '[PASS]' 'Python package fastapi available' } else { Add-Result '[WARN]' 'fastapi not importable' }`n    } catch { Add-Result '[WARN]' 'fastapi import check failed' }`n    # 8. Expected scripts existence`n    $scripts = @('scripts/dev/dev.ps1','package.json','pnpm-lock.yaml')`n    foreach ($s in $scripts) {`n        if (Test-Path $s) { Add-Result '[PASS]' ( 'Script/file exists: ' + $s ) } else { Add-Result '[FAIL]' ( 'Missing expected script/file: ' + $s ) }`n    }`n    # 9. Port checks (common dev ports: 8000 API, 5173 Vite)`n    $ports = @(8000,5173)`n    foreach ($p in $ports) {`n        $tcp = Test-NetConnection -Port $p -WarningAction SilentlyContinue`n        if ($tcp.TcpTestSucceeded) { Add-Result '[PASS]' ( 'Port ' + $p + ' reachable' ) } else { Add-Result '[WARN]' ( 'Port ' + $p + ' not reachable (service may be stopped)' ) }`n    }`n    # 10. Diagnostic directory writability`n    $diagDir = Join-Path -Path $PSScriptRoot -ChildPath '..' -Resolve | Join-Path -ChildPath '.runtime/diagnostics'`n    try {`n        New-Item -ItemType Directory -Path $diagDir -Force | Out-Null`n        $testFile = Join-Path $diagDir 'write_test.tmp'`n        Set-Content -Path $testFile -Value 'test' -Force`n        Remove-Item $testFile -Force`n        Add-Result '[PASS]' 'Diagnostic directory writable'`n    } catch { Add-Result '[FAIL]' 'Diagnostic directory not writable' }`n    # Output summary`n    foreach ($r in $results) { Write-Output ($r.Status + ' ' + $r.Message) }`n    # Determine overall status`n    $hasFail = $results | Where-Object { $_.Status -eq '[FAIL]' }`n    if ($hasFail) { Write-Output '`nNext action: Resolve the first failure above.'; exit 1 } else { Write-Output '`nAll critical checks passed. Recommended next step: proceed with dev start.'; exit 0 }`n}`n`nswitch ($Command) {`n    'doctor' { Invoke-Doctor }`n    default { Write-Output ('Unsupported command: ' + $Command) }`n}`n
